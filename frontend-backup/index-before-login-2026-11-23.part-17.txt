function calcMacdTouchSeries(series){

  const tr=Math.max(c[i].high-c[i].low,Math.abs(c[i].high-c[i-1].close),Math.abs(c[i].low-c[i-1].close));

 for(let i=2;i<c.length;i++){
  if(![m.macd[i],m.signal[i],m.macd[i-1],m.signal[i-1],gaps[i],atr[i]].every(Number.isFinite)){inTouch=false;lastTouchDirection=null;continue;}
  const tolerance=tolAt(i),touch=gaps[i]<=tolerance;
  if(!touch){inTouch=false;lastTouchDirection=null;continue;}
  const macdSlope=m.macd[i]-m.macd[i-1],signalSlope=m.signal[i]-m.signal[i-1];
  const direction=macdSlope>0&&signalSlope>0?'buy':macdSlope<0&&signalSlope<0?'sell':null;
  if(!direction){inTouch=true;continue;}
  if(inTouch&&lastTouchDirection===direction)continue;
  const bullishTouch=direction==='buy',bearishTouch=direction==='sell';
  const item={signalPrice:c[i].close,signalTime:c[i].closeTime,candleTime:c[i].time,macd:m.macd[i],signal:m.signal[i],hist:m.hist[i],gap:gaps[i],tolerance,atr:atr[i],direction,reason:bullishTouch?'MACD 3D encostou na linha de sinal com inclinação ascendente':'MACD 3D encostou na linha de sinal com inclinação descendente'};
  events[i]=item;
  if(bullishTouch)buySignals.set(c[i].closeTime,item);
  if(bearishTouch)sellSignals.set(c[i].closeTime,item);
  inTouch=true;lastTouchDirection=direction;
 }
 return {candles:c,macd:m,atr,gaps,buySignals,sellSignals,events};
}

function runWeeklyConfluenceBacktest(raw){
 const c=(raw||[]).filter(x=>Number.isFinite(x.time)&&Number.isFinite(x.closeTime)&&x.closeTime<Date.now()&&[x.open,x.high,x.low,x.close].every(Number.isFinite)).sort((a,b)=>a.time-b.time);
 const buySignals=new Map(),sellSignals=new Map(),trades=[],fee=0.001;
 if(c.length<205)return {ready:false,reason:'Histórico semanal insuficiente',n:c.length,buySignals,sellSignals,trades,period:'1w',current:{ready:false,label:'AGUARDAR',reasons:['Histórico semanal insuficiente'],buy:false,sell:false}};
 const close=c.map(x=>x.close),m=calcMACDSeries(close),rv=rsi(close),kd=calcKDJSeries(c),smaN=(a,n)=>a.map((_,i)=>i+1<n?null:a.slice(i+1-n,i+1).reduce((x,y)=>x+y,0)/n);
 const bandSma=smaN(close,20),bandEma=ema(close,21),sma200=smaN(close,200),buyAt=Array(c.length).fill(false),sellAt=Array(c.length).fill(false);
 const readyAt=i=>[m.macd[i],m.signal[i],m.hist[i],rv[i],rv[i-1],kd.k[i],kd.d[i],kd.k[i-1],kd.d[i-1],bandSma[i],bandEma[i],sma200[i]].every(Number.isFinite);
 for(let i=2;i<c.length;i++){
  if(!readyAt(i)||!readyAt(i-1))continue;
  const macdCrossUp=m.macd[i-1]<=m.signal[i-1]&&m.macd[i]>m.signal[i],macdCrossDown=m.macd[i-1]>=m.signal[i-1]&&m.macd[i]<m.signal[i];
  const macdBull=m.hist[i]>0&&(macdCrossUp||(m.hist[i]>m.hist[i-1]&&m.hist[i-1]>=m.hist[i-2]&&m.macd[i]>m.signal[i]));
  const macdBear=m.hist[i]<0&&(macdCrossDown||(m.hist[i]<m.hist[i-1]&&m.hist[i-1]<=m.hist[i-2]&&m.macd[i]<m.signal[i]));
  const rsiBull=(rv[i]>=50&&rv[i]>rv[i-1])||(rv[i]>=40&&rv[i]>rv[i-1]&&rv[i-1]<40);
  const rsiBear=(rv[i]<50&&rv[i]<rv[i-1])||(rv[i]<60&&rv[i]<rv[i-1]&&rv[i-1]>=60);
  const kdjCrossUp=kd.k[i-1]<=kd.d[i-1]&&kd.k[i]>kd.d[i],kdjCrossDown=kd.k[i-1]>=kd.d[i-1]&&kd.k[i]<kd.d[i];
  const kdjBull=(kdjCrossUp&&kd.k[i]<=65)||(kd.k[i]>kd.d[i]&&kd.k[i]<80&&kd.k[i]>=kd.k[i-1]);
  const kdjBear=(kdjCrossDown&&kd.k[i]>=35)||(kd.k[i]<kd.d[i]&&kd.k[i]>20&&kd.k[i]<=kd.k[i-1]);
  const bandLow=Math.min(bandSma[i],bandEma[i]),bandHigh=Math.max(bandSma[i],bandEma[i]);
  const nearBand=c[i].close>=bandLow*0.90,aboveBand=c[i].close>=bandLow;
  const recovery=rv[i]>=40&&rv[i]>rv[i-1]&&m.hist[i]>m.hist[i-1]&&c[i].close>=sma200[i]*0.82;
  const breakdown=c[i].close<bandHigh&&c[i].close<sma200[i]&&m.hist[i]<m.hist[i-1];
  const buy=macdBull&&rsiBull&&kdjBull&&(aboveBand||nearBand&&recovery);
  const sell=macdBear&&rsiBear&&kdjBear&&(c[i].close<bandHigh||breakdown);
  const prevBuy=i>2&&buyAt[i-1],prevSell=i>2&&sellAt[i-1];
  buyAt[i]=buy;sellAt[i]=sell;
  if(buy&&!prevBuy)buySignals.set(c[i].time,{signalPrice:c[i].close,executionTime:c[i+1]?.time??c[i].closeTime,executionPrice:c[i+1]?.open??c[i].close,reason:'CONFLUÊNCIA SEMANAL MACD + RSI + KDJ + SUPORTE',macd:m.macd[i],signal:m.signal[i],hist:m.hist[i],rsi:rv[i],k:kd.k[i],d:kd.d[i],bandSma:bandSma[i],bandEma:bandEma[i],sma200:sma200[i]});
  if(sell&&!prevSell)sellSignals.set(c[i].time,{signalPrice:c[i].close,executionTime:c[i+1]?.time??c[i].closeTime,executionPrice:c[i+1]?.open??c[i].close,reason:'CONFLUÊNCIA SEMANAL MACD + RSI + KDJ + PERDA DE SUPORTE',macd:m.macd[i],signal:m.signal[i],hist:m.hist[i],rsi:rv[i],k:kd.k[i],d:kd.d[i],bandSma:bandSma[i],bandEma:bandEma[i],sma200:sma200[i]});
 }
 let pos=null,equity=1,peak=1,maxDD=0;
 for(let i=1;i<c.length-1;i++){
  if(sellAt[i]&&pos){const px=c[i+1].open,net=px*(1-fee)/(pos.entry*(1+fee))-1;trades.push({...pos,exitTime:c[i+1].time,exit:px,netReturn:net,exitReason:'Confluência semanal de baixa'});equity*=1+net;peak=Math.max(peak,equity);maxDD=Math.max(maxDD,(peak-equity)/peak);pos=null;}
  else if(buyAt[i]&&!pos)pos={entry:c[i+1].open,entryTime:c[i+1].time,signalTime:c[i].time,entryReason:'Confluência semanal de alta'};
 }
 if(pos){const last=c.at(-1),net=last.close*(1-fee)/(pos.entry*(1+fee))-1;trades.push({...pos,exitTime:last.time,exit:last.close,netReturn:net,exitReason:'Marcação no fim do histórico',forcedClose:true});equity*=1+net;peak=Math.max(peak,equity);maxDD=Math.max(maxDD,(peak-equity)/peak);}
 const i=c.length-1,valid=readyAt(i),recentBuy=buySignals.has(c[i].time),recentSell=sellSignals.has(c[i].time),wins=trades.filter(t=>t.netReturn>0),losses=trades.filter(t=>t.netReturn<=0),grossWin=wins.reduce((a,t)=>a+t.netReturn,0),grossLoss=-losses.reduce((a,t)=>a+t.netReturn,0);
 const current={ready:valid,period:'1w',label:recentBuy?'COMPRA · CONFLUÊNCIA SEMANAL':recentSell?'VENDA · CONFLUÊNCIA SEMANAL':'AGUARDAR · SEM NOVA CONFLUÊNCIA',buy:recentBuy,sell:recentSell,score:recentBuy?100:recentSell?0:50,rsi:rv[i],macd:{macd:m.macd[i],signal:m.signal[i],hist:m.hist[i]},k:kd.k[i],d:kd.d[i],sma200:sma200[i],bandSma:bandSma[i],bandEma:bandEma[i],close:close[i],reasons:[recentBuy?'MACD, RSI e KDJ confirmam alta; preço avaliado em relação à Bull Market Support Band e SMA 200 semanal':recentSell?'MACD, RSI e KDJ confirmam baixa; perda de suporte semanal verificada':'Sem nova confluência completa no último candle semanal fechado.']};
 return {ready:true,n:c.length,start:c[0].time,end:c[i].time,buySignals,sellSignals,trades,fee,winRate:trades.length?wins.length/trades.length*100:0,totalReturn:(equity-1)*100,buyHold:(close[i]/close[0]-1)*100,maxDD:maxDD*100,profitFactor:grossLoss?grossWin/grossLoss:null,current,period:'1w'};
}
