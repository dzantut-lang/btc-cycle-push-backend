 function simulate(begin,end,variant){

  const benchmark=(1-cost)*(lastClose/firstOpen)*(1-cost);
  function closeAt(price){
   const net=(1-cost)*(price/entryPrice)*(1-cost)-1;
   closed++;sumTradeNet+=net;

    trs.push(Math.max(c.high-c.low,Math.abs(c.high-prev.close),Math.abs(c.low-prev.close)));
   }

  }

   }

  }
