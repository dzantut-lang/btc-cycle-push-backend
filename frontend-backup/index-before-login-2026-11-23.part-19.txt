function calcKDJSeries(series){

function runDailyMacdEmaBacktest(raw,period='1d'){
 const sig=calcDailyMacdEmaSignals(raw,period),c=sig.candles||[],fee=0.001,slippage=0.0005,trades=[];
 let pos=null,equity=1,peak=1,maxDD=0;
 const mark=value=>{if(Number.isFinite(value)&&value>0){peak=Math.max(peak,value);maxDD=Math.max(maxDD,(peak-value)/peak);}};
 if(!sig.ready||c.length<206)return {ready:false,reason:'Histórico insuficiente para candles '+(period==='3d'?'3D':'diários'),trades,buyCount:sig.buySignals.size,sellCount:sig.sellSignals.size,winRate:null,totalReturn:null,maxDD:null,buyHold:null,closedTrades:0,openPosition:false,comparisonPeriod:null};
 // A comparação começa no primeiro candle em que a EMA 200 e o MACD estão disponíveis.
 const firstValidIndex=Array.isArray(sig.e200)?sig.e200.findIndex(Number.isFinite):-1;
 const startIndex=Math.max(0,firstValidIndex);
 const entryIndex=startIndex+1;
 if(entryIndex>=c.length||!Number.isFinite(c[entryIndex].open)||c[entryIndex].open<=0)return {ready:false,reason:'Não há candle seguinte válido para comparar as estratégias',trades,buyCount:sig.buySignals.size,sellCount:sig.sellSignals.size,winRate:null,totalReturn:null,maxDD:null,buyHold:null,closedTrades:0,openPosition:false,comparisonPeriod:null};
 for(let i=startIndex;i<c.length;i++){
  const cur=c[i],next=c[i+1],buy=sig.buySignals.get(cur.closeTime),sell=sig.sellSignals.get(cur.closeTime);
  if(pos){const markedLow=Number.isFinite(cur.low)&&cur.low>0?equity*(cur.low/pos.entry)*(1-fee)*(1-fee)*(1-slippage)*(1-slippage):NaN;mark(markedLow);const marked=equity*(cur.close/pos.entry)*(1-fee)*(1-fee)*(1-slippage)*(1-slippage);mark(marked);}else mark(equity);
  if(!next)break;
  if(pos&&sell){
   const exitPrice=next.open*(1-slippage),net=(exitPrice/pos.entry)*(1-fee)*(1-fee)-1;
   trades.push({entry:pos.entry,exit:exitPrice,entryTime:pos.time,exitTime:next.time,netReturn:net});
   equity*=1+net;pos=null;mark(equity);
  }
  if(!pos&&buy&&Number.isFinite(next.open)&&next.open>0)pos={entry:next.open*(1+slippage),time:next.time};
 }
 if(pos){
  const last=c[c.length-1],exitPrice=last.close*(1-slippage),net=(exitPrice/pos.entry)*(1-fee)*(1-fee)-1;
  trades.push({entry:pos.entry,exit:exitPrice,entryTime:pos.time,exitTime:last.closeTime,netReturn:net,openPosition:true});
  equity*=1+net;mark(equity);
 }
 const closed=trades.filter(t=>!t.openPosition),wins=closed.filter(t=>t.netReturn>0).length;
 let realizedEquity=1;closed.forEach(t=>{realizedEquity*=1+t.netReturn;});
 const realizedReturn=(realizedEquity-1)*100;
 const winRate=closed.length?100*wins/closed.length:0,totalReturn=(equity-1)*100;
 // Buy-and-hold uses the same start date/open, end close, and entry/exit fee assumptions.
 const buyHold=((c[c.length-1].close*(1-slippage)/(c[entryIndex].open*(1+slippage)))*(1-fee)*(1-fee)-1)*100;
 return {ready:true,trades,closedTrades:closed.length,openPosition:trades.some(t=>t.openPosition),buyCount:sig.buySignals.size,sellCount:sig.sellSignals.size,winRate,realizedReturn,totalReturn,maxDD:maxDD*100,buyHold,comparisonPeriod:{startTime:c[entryIndex].time||c[entryIndex].closeTime,endTime:c[c.length-1].closeTime,startPrice:c[entryIndex].open,endPrice:c[c.length-1].close,feePerSidePct:fee*100,slippagePerSidePct:slippage*100,candleCount:c.length-entryIndex}};
}
